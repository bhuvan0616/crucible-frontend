## SDK Integration

**MedusaJS SDK order retrieve:**
```typescript
sdk.store.order.retrieve(id: string, params?: {
  fields?: string;
  expand?: string;
})
```

Use `fields: '*'` to get all order fields including `items`, `shipping_address`, `billing_address`, `payment_methods`, `totals`.

Response shape:
```typescript
{
  order: {
    id: string;
    display_id?: number;
    status: string;
    created_at: string;
    items: OrderLineItem[];
    shipping_address: Address | null;
    billing_address: Address | null;
    payment_methods: PaymentMethod[];
    subtotal: number;
    shipping_total: number;
    tax_total: number;
    total: number;
  }
}
```

## Data Types

```typescript
interface OrderLineItem {
  id: string;
  title: string;           // product name
  variant?: {
    title: string;         // e.g., "Color: Red"
    options?: { title: string }[];
  };
  quantity: number;
  unit_price: number;       // in paise — use parsePaise()
  metadata?: {
    customization?: string; // engraving text
  };
}

interface Address {
  first_name?: string;
  last_name?: string;
  address_1?: string;
  address_2?: string;
  city?: string;
  province?: string;
  postal_code?: string;
  country?: string;
  phone?: string;
}

interface PaymentMethod {
  provider_id: string;      // e.g., "stripe"
  data?: {
    last4?: string;
    brand?: string;
  };
}
```

**Props:** `orderId: string`

**State:**
- `order: OrderDetail | null` — full order from Medusa
- `isLoading: boolean` — skeleton visibility
- `error: string | null` — error message

**Fetching:**
```typescript
useEffect(() => {
  if (!isAuthenticated) return;
  const fetchOrder = async () => {
    setIsLoading(true);
    setError(null);
    try {
      const { order } = await sdk.store.order.retrieve(orderId, { fields: '*' });
      setOrder(order as unknown as OrderDetail);
    } catch (err: any) {
      setError(err.message || "Failed to load order");
    } finally {
      setIsLoading(false);
    }
  };
  fetchOrder();
}, [orderId, isAuthenticated]);
```
