```typescript
const handlePlaceOrder = async () => {
  setIsProcessing(true)
  try {
    const cartId = useCartStore.getState().cartId
    if (!cartId) throw new Error("No cart")
    
    const { type, order, error } = await sdk.store.cart.complete(cartId)
    
    if (type === "order") {
      setOrderData(order)
      setCurrentStep(CheckoutStep.Confirmation)
      // Clear cart
      localStorage.removeItem("cart_id")
      useCartStore.getState().clearCart()
    } else {
      throw new Error(error?.message || "Order failed")
    }
  } catch (error: any) {
    alert(error.message || "Failed to place order")
  } finally {
    setIsProcessing(false)
  }
}
```

**Layout:** Two-column layout (form on left, order summary on right) with step indicator at top

2. Update clearCart to optionally reset completely vs create new cart:

```typescript
clearCart: async (resetOnly = false) => {
  if (resetOnly) {
    // Complete reset for post-checkout (don't create new cart)
    localStorage.removeItem("cart_id")
    set({
      cartId: null,
      items: [],
      totals: { subtotal: 0, shipping_total: 0, tax_total: 0, total: 0 },
      isLoading: false,
      isInitialized: false, // Force re-init on next access
    })
    return
  }
  // Original: create new cart
  ...
}
```

- Removing cart_id from localStorage
- Calling useCartStore.getState().clearCart() after order success
