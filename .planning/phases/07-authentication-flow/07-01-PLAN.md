**login action:**
```typescript
login: async (email: string, password: string) => {
  set({ isLoading: true, error: null })
  try {
    const { authToken, location } = await sdk.auth.login(
      "customer",
      "emailpass",
      { email, password }
    )
    const { user } = await sdk.auth.getSession()
    set({ isAuthenticated: true, user, isLoading: false })
  } catch (error: any) {
    set({ error: error.message || "Login failed", isLoading: false })
    throw error
  }
}
```

**register action:**
```typescript
register: async (data: RegisterInput) => {
  set({ isLoading: true, error: null })
  try {
    const { authToken } = await sdk.auth.register(
      "customer",
      "emailpass",
      data
    )
    const { user } = await sdk.auth.getSession()
    set({ isAuthenticated: true, user, isLoading: false })
  } catch (error: any) {
    const message = error.message?.includes("already exists")
      ? "Email already registered"
      : error.message || "Registration failed"
    set({ error: message, isLoading: false })
    throw error
  }
}
```

**logout action:**
```typescript
logout: async () => {
  await sdk.auth.logout()
  set({ isAuthenticated: false, user: null, error: null })
}
```

**checkAuth action:**
```typescript
checkAuth: async () => {
  set({ isLoading: true })
  try {
    const isAuth = await sdk.auth.isAuthenticated()
    if (isAuth) {
      const { user } = await sdk.auth.getSession()
      set({ isAuthenticated: true, user, isLoading: false })
    } else {
      set({ isAuthenticated: false, user: null, isLoading: false })
    }
  } catch {
    set({ isAuthenticated: false, user: null, isLoading: false })
  }
}
```

**clearError action:**
```typescript
clearError: () => set({ error: null })
```

**Persist config:** No persistence needed — JWT stored by SDK, checkAuth on init restores state

**Export:** `useAuthStore` as default export
**Also export:** `selectIsAuthenticated`, `selectUser` selectors
